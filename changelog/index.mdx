import { Update } from '@astrojs/starlight/components';

<Update label="July 21, 2026" features={[<u/>]}>
  * Fixed high-priority bug in `tasks assign` help and fixed cli predefined alias bug: `hypertask task assign --help` now consistently documents `--user <id>` instead of the incorrect `--assignee <id>` flag; `tasks unassign` was broken/incompatible with the same intent and is now fixed to accept `--assignee <id>` (HTPR-4109)
  * Fixed `tasks create` ignoring invalid `--parent-task` values: now validates that the value is a valid positive integer and reports clear validation errors before attempting to create or update a parentless task, preventing parent missing or invalid relationships (HTPR-4260)
  * Fixed `task create` / `task update` `--estimate` help text and validation: help now clearly lists size indices `2`–`6` for XS–XL sizing and validates against valid values before the API call; estimates outside the accepted range now trigger a clear error instead of being silently dropped or incorrectly processed (HTPR-4450)
</Update>

<Update label="July 21, 2026" features={[<u/>]}>
  * Fixed comment emails for agents: Comments created by agents, MCP tools, and CLI now send email notifications to followers and owners, matching web-UI behavior ([HTPR-4158](https://app.hypertask.ai/detail/project-15/4158))
</Update>

<Update label="July 21, 2026" features={[<u/>]}>
  * Improved /chat board filter: choose "All boards" to see chats from every board or filter to a single board's chats close the default ([HTPR-4483](https://app.hypertask.ai/detail/project-15/4483))
</Update>

<Update label="July 21, 2026" features={[<u/>]}>
  * Improved session length: Choose login sessions of 7 or 30 days instead of the previous 7-day default ([HTPR-4489](https://app.hypertask.ai/detail/project-15/4489))
  * Added per-chat URLs: Each conversation in AI Chat gets its own shareable link that deep-loads that specific conversation when visited ([HTPR-4485](https://app.hypertask.ai/detail/project-15/4485))
  * Improved table view readability: Task titles now wrap by default on smaller screens with a Ctrl+K toggle to switch to truncation mode ([HTPR-4233](https://app.hypertask.ai/detail/project-15/4233))
</Update>

<Update label="July 20, 2026" features={[<u/>]}>
  * Fixed no boards empty state: Boardless accounts now show a clear "No boards yet" screen instead of dead-ending on a blank screen (HTPR-4206)
</Update>

<Update label="July 14, 2026" features={[<u/>]}>
  * Improved AI chat context: AI now knows which View you're on and can switch between Views via chat, CLI, API, and MCP (HTPR-4294)
</Update>
